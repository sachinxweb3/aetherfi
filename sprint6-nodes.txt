"use client";

import { Handle, Position } from "@xyflow/react";
import { motion } from "framer-motion";
import { ReactNode } from "react";

import { FlowNodeData } from "./types/FlowNodeData";

interface BaseNodeProps {
  data: FlowNodeData;
  icon: ReactNode;
  accent: string;
}

export default function BaseNode({
  data,
  icon,
  accent,
}: BaseNodeProps) {
  return (
    <motion.div
      whileHover={{
        y: -4,
        scale: 1.02,
      }}
      transition={{
        duration: 0.2,
      }}
      className="group relative min-w-[240px]"
    >
      {/* Glow */}
      <div
        className="absolute -inset-1 rounded-3xl opacity-0 blur-xl transition-opacity duration-300 group-hover:opacity-100"
        style={{
          background: `linear-gradient(135deg, ${accent}40, transparent)`,
        }}
      />

      {/* Card */}
      <div className="relative rounded-3xl border border-white/10 bg-white/80 p-5 shadow-xl backdrop-blur-2xl dark:bg-neutral-900/80">
        <Handle
          type="target"
          position={Position.Top}
          className="!h-3 !w-3 !border-2 !border-white !bg-blue-500"
        />

        <div className="flex items-center gap-4">
          <div
            className="flex h-12 w-12 items-center justify-center rounded-2xl"
            style={{
              background: `${accent}20`,
              color: accent,
            }}
          >
            {icon}
          </div>

          <div className="flex-1">
            <h3 className="text-base font-semibold">
              {data.title}
            </h3>

            {data.subtitle && (
              <p className="mt-1 text-sm text-muted-foreground">
                {data.subtitle}
              </p>
            )}
          </div>
        </div>

        <Handle
          type="source"
          position={Position.Bottom}
          className="!h-3 !w-3 !border-2 !border-white !bg-blue-500"
        />
      </div>
    </motion.div>
  );
}
